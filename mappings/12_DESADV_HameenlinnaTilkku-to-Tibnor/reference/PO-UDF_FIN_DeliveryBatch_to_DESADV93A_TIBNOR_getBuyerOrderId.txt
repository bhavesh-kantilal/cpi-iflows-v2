public String getBuyerOrderId(String orderId, Container container) throws StreamTransformationException{
    String returnValue = orderId;
    int pos = 0;

    if(orderId.contains("-")) {
        pos = orderId.indexOf("-");
        returnValue = orderId.substring(0, pos);
    }

    return returnValue;
}
